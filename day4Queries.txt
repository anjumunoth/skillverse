  -- queryHash
  	-- stored as part of metadata of collection
    -- stores the db, collection name and commands for the various queries executed
    
  -- planCache
Cache 
	-- part of RAM memory
  -- Used to store frequently accessed items
  -- Faster access
  -- Eviction policy
  	-- cache size -- limited 
    	-- not able to keep the entire data in the cache
    -- read 
    	-- check if required data is in the cache 
      	-- if present in the cache -- cache hit
        	-- return from the cache itself the required data
        -- if not present in the cache -- cache miss
        	-- read from the disk ; store in the cache(if cache memory not available -- will do eviction); return the result to user
        
Mongodb 
	-- wired tiger cache
  	-- cache docs, index entries
    -- of a configurable size
    -- read query
    	-- match the query shape with query hash already present
      -- if matches -- execute based on corresponding plan cache
      -- if no match -- process of execution, get the winning plan, store it in a new query hash and plan cache and return the result
      
Execution Plan -- picked from the cache
      
  
Whenever a read query comes in -- Process in which it is executed
	-- Match the queryHash and check if this query matches any of the queries already executed 
  	-- If there ia a match -- return the result from the plan cache -- not excute the command, fetch from the cache
    -- If there is no match 
    	-- find all the candidate plans -- run all the candidate plans -- whichever is the fastest at the first stage -- go ahead mark it as winning plan and stop the others and add them to the rejected plan
    	-- add the winning plan as the plan cache for this query shape

Compound Index 
	-- Index on multiple fields
  -- Important rules
  	-- Position of fields is very important
    -- Sort order of fields is very important
  -- Prefix compression is done by default
    
    
db.zipcode.createIndex({state:1,city:-1})


db.zipcode.find({state:"MA"});// Compound index will be used
db.zipcode.find({city:"AGAWAM"});// Compound index will not be used

db.zipcode.find({}).sort(state:1);// Compound index will be used
db.zipcode.find({}).sort(state:-1);// Compound index will be used
db.zipcode.find({}).sort(city:1);// Compound index will not be used
db.zipcode.find({}).sort(city:-1);// Compound index will not be used

db.zipcode.find({}).sort(state:1,city:1);// Compound index will not be used
db.zipcode.find({}).sort(state:1,city:-1);// Compound index will be used
db.zipcode.find({}).sort(state:-1,city:1);// Compound index will be used
db.zipcode.find({}).sort(state:-1,city:-1);// Compound index will not be used

db.zipcode.find({city:"AGAWAM"}).sort({state:1});// Compound index will be used
// Plan : Use the compound index -- get the sorted state wise docs ; fetch -- filter  -- winning plan
// Plan : COLL SCAN 

db.zipcode.find({city:"AGAWAM"}).sort({state:-1});// Compound index will be used
// Plan : Use the compound index -- get the sorted state wise docs in desc order ; fetch -- filter
