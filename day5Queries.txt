Partial filter expression
	-- Filter conditions supports only certain operations
db.employee.createIndex({salary:-1},{name:"partialSalaryWithSalaryCheck",partialFilterExpression:{salary :{$gt:5000}}})
// Will index all the docs based on salary which have a salary > 5000

db.employee.createIndex({salary:-1},{partialFilterExpression:{empName :/a$/i, name:"partialSalaryWithNameCheck"}})
// Will index all the docs based on the salary where the doc has an empName ending with "a" -- Index will not be created becaues regex are not supported

db.employee.createIndex({salary:-1},{partialFilterExpression:{empName :"tara", name:"partialSalaryWithNameAsTara"}})
// Will index only those 5/13 docs with name as "tara"
// Binary tree index entries : based on salary;
//


db.employee.find({salary:{$gt:10000}}); // partialSalaryWithSalaryCheck
db.employee.find({salary:{$gt:1000}}); // COLLSCAN
// Plan 1: Collscan
// plan 2: try to fetct >5000 from index and then 1000 to 5000 by doing collscan
// plan 2 -- rejected plan


db.employee.find({salary:{$lt:10000}}); // COLLSCAN


db.employee.find({empName:"tara"});// Query doesnot have the indexed field -- No partial index usage
db.employee.find({empName:{/a$/i}});// Query doesnot have the indexed field -- No partial index usage

db.employee.find({salary:{$gt:10000}});// Not use this index:partialSalaryWithNameAsTara
db.employee.find({salary:{$gt:10000},empName:"tara"});// partialSalaryWithNameAsTara
db.employee.find({salary:{$gt:10000},empName:/a$/});// Will not use the index because there can be other docs like sita which ends with a and which may be in the index

db.collName.getIndexes()
db.collName.dropIndex({name:"partialSalaryWithNameAsTara"})



TTL Index :
	--Total time to live index
  -- Index can delete docs
  	-- Delete docs for which the expiresIn time in seconds has expired from the indexed field
  -- TTL index can be created only on date data type fields or arrays which have dates
  -- Usecases
  	-- Automatically delete some docs after a period of time
    -- Session expiration
    -- Logs rotation
  
  
db.bankingSession.insertMany([
	{sessionId:101,timeStart:new Date(2026,9,05)}
  {sessionId:102,timeStart:new Date(2025,2,2)}
  {sessionId:103,timeStart:new Date(2026,2,2)}
  {sessionId:104,timeStart:new Date(2027,2,2)}
  {sessionId:105,timeStart:new Date(2026,9,05,17)}
])

db.bankingSession.createIndex({timeStart:1},{expiresIn:300})

Hashed Index :
	db.employee.createIndex({employeeId:"hashed"})
  -- Simple index or compound index
  -- Can create only using shell
  -- Mainly used with sharding
  -- Usually created on fiels which change monotonically like ObjectId or timestamp 
  -- Try to evenly distribute the data 
  -- Work
  	-- indexed field value --> sent to a hash algorithm -- give a hashed value of this indexed field -- hashed value becomes the index entry
    -- Query which comes on the hashed field --> convert the query' condition to a hash value -- hash values will be compared -- based on the result set will be formed
